using UnityEngine;
using UnityEngine.UI;
namespace WoolGame
{
    // Sprite.OverrideGeometry is not retained by standalone Sprite .asset files.
    // Reapply the cutout after loading, before the Image builds its UI mesh.
    [RequireComponent(typeof(Image))]
    public sealed class WoolSpriteSilhouette : MonoBehaviour
    {
        public enum Shape { Circle, Pointer, Unlock }
        public Shape shape;
        void OnEnable(){Apply();}
        public void Apply()
        {
            var image=GetComponent<Image>();if(!image.sprite)return;
            var s=image.sprite;float w=s.rect.width/s.pixelsPerUnit,h=s.rect.height/s.pixelsPerUnit;
            if(shape==Shape.Circle)
            {
                var v=new Vector2[65];var t=new ushort[192];
                for(int i=0;i<64;i++){float a=i*Mathf.PI*2/64;v[i+1]=new Vector2(Mathf.Cos(a)*w,Mathf.Sin(a)*h)*.485f;t[i*3]=0;t[i*3+1]=(ushort)(i+1);t[i*3+2]=(ushort)((i+1)%64+1);}
                s.OverrideGeometry(v,t);
            }
            else if(shape==Shape.Pointer)s.OverrideGeometry(new[]{new Vector2(-w*.475f,h*.405f),new Vector2(w*.475f,h*.405f),new Vector2(w*.45f,h*.19f),new Vector2(w*.0875f,-h*.464f),new Vector2(-w*.0875f,-h*.464f),new Vector2(-w*.45f,h*.19f)},new ushort[]{0,1,2,0,2,3,0,3,4,0,4,5});
            // The unlock silhouette uses a rounded rectangle crop; its cream edge matches the prize disk.
            image.type=Image.Type.Simple;image.useSpriteMesh=true;image.SetVerticesDirty();
        }
    }
}
